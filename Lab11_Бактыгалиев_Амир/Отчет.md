```jsx
import React, { useState } from 'react';

export function ThemeSwitch({ themes, onThemeChange }) {
  const [selectedTheme, setSelectedTheme] = useState(themes[0]?.id || 'dark');

  const handleSelect = (themeId) => {
    setSelectedTheme(themeId);
    if (onThemeChange) onThemeChange(themeId);
  };

  return (
    <div className="theme-switch">
      <span className="theme-title">Текущая тема: <strong>{selectedTheme}</strong></span>
      <div className="theme-buttons">
        {themes.map(t => (
          <button
            key={t.id}
            onClick={() => handleSelect(t.id)}
            className={selectedTheme === t.id ? 'active' : ''}
          >
            {t.icon} {t.name}
          </button>
        ))}
      </div>
    </div>
  );
}
```

```jsx
import React, { useState } from 'react';

export function ManualTimer({ label, initialMinutes = 15, step = 5 }) {
  const [minutes, setMinutes] = useState(initialMinutes);

  const increment = () => setMinutes(m => m + step);
  const decrement = () => setMinutes(m => Math.max(0, m - step));
  const reset = () => setMinutes(initialMinutes);

  return (
    <div className="timer-box">
      <h3>{label}</h3>
      <div className="timer-display">{minutes} мин.</div>
      <div className="timer-actions">
        <button onClick={decrement}>-{step} мин</button>
        <button onClick={increment}>+{step} мин</button>
        <button onClick={reset}>Сброс</button>
      </div>
    </div>
  );
}
```

```jsx
// src/components/ProductCard.jsx
import React, { useState } from 'react';

export function ProductCard({ title, price, category, onQuantityChanged }) {
  const [count, setCount] = useState(0);

  const increment = () => {
    const next = count + 1;
    setCount(next);
    if (onQuantityChanged) onQuantityChanged(title, next, price);
  };

  const decrement = () => {
    if (count <= 0) return;
    const next = count - 1;
    setCount(next);
    if (onQuantityChanged) onQuantityChanged(title, next, price);
  };

  const reset = () => {
    setCount(0);
    if (onQuantityChanged) onQuantityChanged(title, 0, price);
  };

  return (
    <div className="product-card">
      <div className="category-tag">{category}</div>
      <h4>{title}</h4>
      <div className="price-tag">{price.toLocaleString()} ₽ / шт.</div>
      
      <div className="stepper">
        <button onClick={decrement} disabled={count === 0}>-</button>
        <span className="count-display">{count} шт.</span>
        <button onClick={increment}>+</button>
      </div>

      <div className="subtotal">
        Итого: <strong>{(count * price).toLocaleString()} ₽</strong>
      </div>
    </div>
  );
}
```

```jsx
    if (!inputValue.trim()) return;
    setTasks([...tasks, { id: Date.now(), title: inputValue.trim(), completed: false }]);
    setInputValue('');
  };

  const toggleTask = (id) => {
    setTasks(tasks.map(t => t.id === id ? { ...t, completed: !t.completed } : t));
  };

  const deleteTask = (id) => {
    setTasks(tasks.filter(t => t.id !== id));
  };

  return (
    <div className="todo-container">
      <form onSubmit={addTask} className="input-group">
        <input
          type="text"
          value={inputValue}
          onChange={e => setInputValue(e.target.value)}
          placeholder="Новая учебная задача..."
        />
        <button type="submit">Добавить</button>
      </form>

      <ul className="tasks-list">
        {tasks.map(task => (
          <TodoItem
            key={task.id}
            task={task}
            onToggle={toggleTask}
            onDelete={deleteTask}
          />
        ))}
      </ul>
    </div>
  );
}
```

## 🧠 4. Ответы на контрольные вопросы


#### Вопрос 1: Что такое компонент в React?
**Ответ:**  
Компонент — это независимая, изолированная и повторно используемая функциональная часть пользовательского интерфейса, инкапсулирующая собственную верстку (JSX) и логику поведения. В современном React компоненты описываются как функции JavaScript, принимающие входные данные (props) и возвращающие структуру Virtual DOM.



#### Вопрос 2: Для чего нужны props?
**Ответ:**  
Props (сокращение от properties) служат для однонаправленной передачи данных и функций обратного вызова (callbacks) от родительского компонента к дочернему. Они позволяют настраивать внешний вид и поведение дочерних компонентов, обеспечивая модульность и повторное использование.



#### Вопрос 3: Почему props считаются входными данными?
**Ответ:**  
Props передаются компоненту извне, аналогично входным параметрам математической функции, и являются строго неизменяемыми (read-only / immutable). Дочерний компонент ни при каких условиях не должен модифицировать полученный объект props, соблюдая принцип чистоты функций.



#### Вопрос 4: Что такое state?
**Ответ:**  
State (состояние) — это внутреннее изменяемое хранилище данных компонента, определяющее его текущий вид и реактивное поведение во времени. Изменение состояния через функцию-сеттер автоматически инициирует повторный рендеринг компонента и его дочерних элементов.



#### Вопрос 5: Как работает хук useState?
**Ответ:**  
Хук useState(initialValue) вызывается на верхнем уровне функционального компонента и возвращает кортеж из двух элементов: текущего значения состояния и функции-сеттера [value, setValue]. При вызове setValue(newValue) или setValue(prev => prev + 1) React сохраняет новое значение и планирует новый цикл рендера.



#### Вопрос 6: Когда происходит повторный рендер (re-render)?
**Ответ:**  
Повторный рендер компонента происходит при: 1) вызове функции изменения локального состояния (useState); 2) изменении значений переданных props от родителя; 3) повторном рендере родительского компонента (если не используется React.memo); 4) изменении значения контекста (useContext), на который подписан компонент.



#### Вопрос 7: Как обработать событие click в React?
**Ответ:**  
Событие обрабатывается через синтетический атрибут onClick, которому передается ссылка на функцию-обработчик: onClick={handleClick} или стрелочная функция onClick={() => setCount(c => c + 1)}. Нельзя передавать немедленный вызов функции onClick={handleClick()}, иначе она выполнится в момент самого рендера.



#### Вопрос 8: Почему нельзя изменять state прямым присваиванием?
**Ответ:**  
Прямое присваивание (например, count = count + 1 или items.push(x)) изменяет значение в памяти, но не уведомляет механизм React. Хук useState не зарегистрирует событие изменения, Virtual DOM не выполнит диффинг (reconciliation), и пользовательский интерфейс останется в устаревшем состоянии.



#### Вопрос 9: Чем локальное состояние отличается от props?
**Ответ:**  
Props передаются компоненту из родителя, являются неизменяемыми для получателя (read-only) и определяют внешнюю конфигурацию. State создается, инициализируется и управляется самим компонентом, является приватным и отражает интерактивную динамику интерфейса.



#### Вопрос 10: Как передать обработчик из родителя ребенку (lifting state up)?
**Ответ:**  
Родительский компонент объявляет функцию-обработчик, которая изменяет состояние родителя, и передает ссылку на нее дочернему компоненту через prop (например: <Child onSelect={handleSelect} />). Дочерний компонент вызывает props.onSelect(data) по клику или вводу, возвращая данные родителю.



#### Вопрос 11: Что такое управляемое поле ввода (controlled input)?
**Ответ:**  
Управляемый компонент — это элемент ввода (<input>, <textarea>, <select>), значение которого берется напрямую из состояния React через атрибут value={val}, а каждое изменение текста отслеживается обработчиком onChange={e => setVal(e.target.value)}. Единственным источником истины является React state.



#### Вопрос 12: Как проверить, что два экземпляра компонента независимы?
**Ответ:**  
В родительском компоненте рендерятся два экземпляра: <Counter title='Счетчик 1' /> и <Counter title='Счетчик 2' />. При нажатии кнопок на первом счетчике изменяется исключительно его значение, а второй счетчик остается на нуле. Это доказывает, что каждый экземпляр обладает собственным изолированным состоянием.
